% \text{三对角矩阵}（\text{上} Hessenberg）
\det \begin{bmatrix}
a_1 & b_1 & 0 & \cdots & 0 \\
c_1 & a_2 & b_2 & \cdots & 0 \\
0 & c_2 & a_3 & \cdots & 0 \\
\vdots & \vdots & \vdots & \ddots & b_{n-1} \\
0 & 0 & 0 & c_{n-1} & a_n
\end{bmatrix} 
= D_n

% \text{三对角矩阵行列式递推公式}
\text{\text{其中} } D_0 = 1, \quad D_1 = a_1, \quad
D_k = a_k D_{k-1} - b_{k-1} c_{k-1} D_{k-2}, \quad k = 2,3,\dots,n
